# carga média do SIN por hora do dia
import pandas as pd

df = pd.read_csv("curva_carga_2025.csv")

curva = (
    df.groupby("hora")["carga_mw"]
      .mean()
      .round(1)
)

curva.idxmax()   # -> 19h
